<?php

namespace Tests\Feature;

use App\Models\Task;
use App\Models\User;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Illuminate\Foundation\Testing\WithFaker;
use Laravel\Sanctum\Sanctum;
use Tests\TestCase;

class TaskApiTest extends TestCase
{
    use RefreshDatabase;

    private User $user;

    protected function setUp(): void
    {
        parent::setUp();
        $this->user = User::factory()->create();
    }

    /* =====================================================
     * Test 1: GET /api/tasks — List authenticated user tasks
     * ===================================================== */
    public function test_authenticated_user_can_list_their_tasks(): void
    {
        Sanctum::actingAs($this->user);

        // Create 5 tasks for this user
        Task::factory()->count(5)->create(['user_id' => $this->user->id]);
        // Create 3 tasks for another user (should NOT appear)
        Task::factory()->count(3)->create();

        $response = $this->getJson('/api/tasks');

        $response->assertOk()
                 ->assertJsonStructure([
                     'data' => [
                         '*' => ['id', 'title', 'status', 'priority', 'due_date', 'created_at', 'updated_at'],
                     ],
                     'meta' => ['total', 'count', 'per_page', 'current_page', 'total_pages'],
                     'links',
                 ])
                 ->assertJsonPath('meta.total', 5);
    }

    /* =====================================================
     * Test 2: POST /api/tasks — Create with valid data → 201
     * ===================================================== */
    public function test_can_create_task_with_valid_data(): void
    {
        Sanctum::actingAs($this->user);

        $payload = [
            'title'       => 'مراجعة الكود',
            'description' => 'مراجعة PR #42 للتأكد من جودة الكود',
            'status'      => 'pending',
            'priority'    => 'high',
            'due_date'    => now()->addDays(3)->format('Y-m-d'),
        ];

        $response = $this->postJson('/api/tasks', $payload);

        $response->assertCreated()
                 ->assertJsonPath('data.title', 'مراجعة الكود')
                 ->assertJsonPath('data.priority', 'high')
                 ->assertJsonPath('message', 'تم إنشاء المهمة بنجاح');

        $this->assertDatabaseHas('tasks', [
            'title'    => 'مراجعة الكود',
            'user_id'  => $this->user->id,
        ]);
    }

    /* =====================================================
     * Test 3: POST /api/tasks — Validation errors → 422
     * ===================================================== */
    public function test_create_task_validation_returns_422(): void
    {
        Sanctum::actingAs($this->user);

        // Missing title (required)
        $response = $this->postJson('/api/tasks', [
            'description' => 'وصف بدون عنوان',
        ]);

        $response->assertUnprocessable()
                 ->assertJsonValidationErrors(['title']);

        // Invalid status
        $response2 = $this->postJson('/api/tasks', [
            'title'  => 'مهمة',
            'status' => 'invalid_status',
        ]);

        $response2->assertUnprocessable()
                  ->assertJsonValidationErrors(['status']);
    }

    /* =====================================================
     * Test 4: PATCH /api/tasks/{task}/toggle — Toggle status
     * ===================================================== */
    public function test_can_toggle_task_status(): void
    {
        Sanctum::actingAs($this->user);

        $task = Task::factory()->create([
            'user_id' => $this->user->id,
            'status'  => 'pending',
        ]);

        // Toggle: pending → done
        $response = $this->patchJson("/api/tasks/{$task->id}/toggle");

        $response->assertOk()
                 ->assertJsonPath('data.status', 'done')
                 ->assertJsonPath('message', 'تم إكمال المهمة');

        // Toggle back: done → pending
        $response2 = $this->patchJson("/api/tasks/{$task->id}/toggle");

        $response2->assertOk()
                  ->assertJsonPath('data.status', 'pending')
                  ->assertJsonPath('message', 'تم إلغاء إكمال المهمة');
    }

    /* =====================================================
     * Test 5: DELETE /api/tasks/{task} — Delete own task
     * ===================================================== */
    public function test_can_delete_own_task_and_cannot_delete_others(): void
    {
        Sanctum::actingAs($this->user);

        $task = Task::factory()->create(['user_id' => $this->user->id]);

        // Delete own task
        $response = $this->deleteJson("/api/tasks/{$task->id}");

        $response->assertOk()
                 ->assertJsonPath('message', 'تم حذف المهمة بنجاح');

        $this->assertDatabaseMissing('tasks', ['id' => $task->id]);

        // Try to delete another user's task
        $otherTask = Task::factory()->create();

        $response2 = $this->deleteJson("/api/tasks/{$otherTask->id}");

        $response2->assertForbidden();
        $this->assertDatabaseHas('tasks', ['id' => $otherTask->id]);
    }
}

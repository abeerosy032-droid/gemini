<?php

namespace App\Http\Requests;

use Illuminate\Contracts\Validation\ValidationRule;
use Illuminate\Foundation\Http\FormRequest;

class StoreTaskRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true; // Sanctum handles auth
    }

    /**
     * @return array<string, ValidationRule|array|string>
     */
    public function rules(): array
    {
        return [
            'title'       => ['required', 'string', 'max:255'],
            'description' => ['nullable', 'string', 'max:2000'],
            'status'      => ['sometimes', 'in:pending,in_progress,done'],
            'priority'    => ['sometimes', 'in:low,medium,high'],
            'due_date'    => ['nullable', 'date', 'after_or_equal:today'],
        ];
    }

    public function messages(): array
    {
        return [
            'title.required'          => 'عنوان المهمة مطلوب',
            'title.max'               => 'العنوان لا يتجاوز 255 حرف',
            'description.max'         => 'الوصف لا يتجاوز 2000 حرف',
            'status.in'               => 'الحالة يجب أن تكون: pending, in_progress, done',
            'priority.in'             => 'الأولوية يجب أن تكون: low, medium, high',
            'due_date.date'           => 'تاريخ الاستحقاق غير صالح',
            'due_date.after_or_equal' => 'تاريخ الاستحقاق يجب أن يكون اليوم أو بعده',
        ];
    }
}
